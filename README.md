CREATE EXTENSION IF NOT EXISTS "pgcrypto";
CREATE EXTENSION IF NOT EXISTS "pg_trgm";  
CREATE EXTENSION IF NOT EXISTS "pgvector";

CREATE TABLE users (
id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
email TEXT UNIQUE,
username TEXT UNIQUE,
name TEXT,
locale TEXT,
timezone TEXT,
created_at TIMESTAMPTZ DEFAULT now(),
last_seen_at TIMESTAMPTZ
);

CREATE TABLE user_facts (
id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
user_id BIGINT NOT NULL,
key TEXT NOT NULL,
value JSONB NOT NULL,
confidence FLOAT DEFAULT 1.0,
source TEXT,
created_at TIMESTAMPTZ DEFAULT now(),
updated_at TIMESTAMPTZ,

UNIQUE (user_id, key)
);

CREATE TABLE user_preferences (
id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
user_id BIGINT NOT NULL,
category TEXT NOT NULL,
value JSONB NOT NULL,
strength FLOAT DEFAULT 1.0,
updated_at TIMESTAMPTZ DEFAULT now(),

UNIQUE (user_id, category)
);

CREATE TABLE user_traits (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id BIGINT NOT NULL,
  trait TEXT NOT NULL,
  score FLOAT CHECK (score >= 0 AND score <= 1),
  source TEXT,
  updated_at TIMESTAMPTZ DEFAULT now(),

  UNIQUE (user_id, trait)
);

CREATE TABLE user_goals (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id BIGINT NOT NULL,
  title TEXT NOT NULL,
  description TEXT,
  status TEXT,
  priority INT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ
);

CREATE TABLE user_habits (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id BIGINT NOT NULL,
  habit TEXT NOT NULL,
  frequency TEXT,
  confidence FLOAT,
  last_observed_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE memories (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id BIGINT NOT NULL,
  title TEXT,
  content TEXT NOT NULL,
  category TEXT,
  emotional_weight INT CHECK (emotional_weight BETWEEN 0 AND 10),
  is_persistent BOOLEAN DEFAULT false,
  valid_until TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ
);

CREATE TABLE emotional_state (
  user_id BIGINT PRIMARY KEY,
  mood TEXT,
  intensity INT CHECK (intensity BETWEEN 0 AND 10),
  updated_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE relationships (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id BIGINT NOT NULL,
  name TEXT NOT NULL,
  role TEXT,
  closeness INT CHECK (closeness BETWEEN 0 AND 10),
  notes TEXT,
  updated_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE user_beliefs (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id BIGINT NOT NULL,
  belief TEXT NOT NULL,
  confidence FLOAT,
  updated_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE embeddings (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id BIGINT NOT NULL,
  source_type TEXT NOT NULL,
  source_id BIGINT,
  vector VECTOR(1536),
  created_at TIMESTAMPTZ DEFAULT now()
);

ALTER TABLE user_habits ADD FOREIGN KEY (user_id) REFERENCES users(id);
ALTER TABLE memories ADD FOREIGN KEY (user_id) REFERENCES users(id);
ALTER TABLE relationships ADD FOREIGN KEY (user_id) REFERENCES users(id);
